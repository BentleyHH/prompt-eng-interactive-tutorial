# UTF-8 erzwingen (verhindert Umlaut-Probleme, falls der Server sonst Latin-1 sendet)
AddDefaultCharset UTF-8

  AddType application/manifest+json .webmanifest
  AddType image/svg+xml .svg
  AddCharset UTF-8 .html
</IfModule>

# Die sichtbare Kopie htaccess.txt nicht ausliefern (falls nach dem Umbenennen liegen geblieben)

</Files>
